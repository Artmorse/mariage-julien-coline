{{- /* Responsive photograph. The source files are 5184 px JPEGs straight out of the camera: they are resized at build time and served as AVIF with a JPEG fallback, with a srcset so that no more pixels than necessary are ever sent to a phone. Parameters: path — path inside assets/ (required) alt — alternative text (required) ratio — optional crop, e.g. "4x5", "3x2" sizes — sizes attribute, defaults to the full content width class — classes applied to the loading — "eager" for the above-the-fold image, "lazy" otherwise */ -}} {{- $path := .path }} {{- $alt := .alt }} {{- $ratio := .ratio }} {{- $sizes := default "(min-width: 768px) 720px, 100vw" .sizes }} {{- $class := default "" .class }} {{- $loading := default "lazy" .loading }} {{- with resources.Get $path }} {{- /* JPEGs out of the camera carry an Exif orientation tag: without AutoOrient first, portrait shots are displayed lying down. */}} {{- $base := images.Filter images.AutoOrient . }} {{- /* Nothing is ever enlarged: a source narrower than the working width is used at its own size, so a small photograph stays sharp instead of being blown up soft. */}} {{- $w := math.Min 1600 $base.Width | int }} {{- /* Optional crop: the "smart" mode keeps the most meaningful area of the photograph. */}} {{- if $ratio }} {{- $parts := split $ratio "x" }} {{- $h := math.Round (mul (div (float $w) (float (index $parts 0))) (float (index $parts 1))) }} {{- $base = $base.Fill (printf "%dx%.0f smart" $w $h) }} {{- else }} {{- $base = $base.Resize (printf "%dx" $w) }} {{- end }} {{- $widths := slice 480 720 1080 1600 }} {{- $jpegs := slice }} {{- $avifs := slice }} {{- range $widths }} {{- if le . $base.Width }} {{- $r := $base.Resize (printf "%dx" .) }} {{- $jpegs = $jpegs | append (printf "%s %dw" $r.RelPermalink $r.Width) }} {{- $a := $base.Resize (printf "%dx avif q65" .) }} {{- $avifs = $avifs | append (printf "%s %dw" $a.RelPermalink $a.Width) }} {{- end }} {{- end }} {{- $fallback := $base.Resize (printf "%dx jpg q80" (math.Min 1080 $base.Width | int)) }} {{- else }} {{- warnf "figure.html: image not found in assets/: %q" $path }} {{- end }}
