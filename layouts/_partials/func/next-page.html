{{- /* Returns the next page in main-menu order, as a dict (Name, URL, Label). The last page points back to the home page. Returns nil when the current page is not in the menu. */ -}} {{- $result := "" }} {{- $menu := site.Menus.main }} {{- $index := -1 }} {{- range $i, $entry := $menu }} {{- if $.IsMenuCurrent "main" $entry }} {{- $index = $i }} {{- end }} {{- end }} {{- if ge $index 0 }} {{- $isLast := eq $index (sub (len $menu) 1) }} {{- $next := index $menu (cond $isLast 0 (add $index 1)) }} {{- $label := cond $isLast "Revenir au début" "Page suivante" }} {{- $result = dict "Name" $next.Name "URL" $next.URL "Label" $label }} {{- end }} {{- return $result }}
